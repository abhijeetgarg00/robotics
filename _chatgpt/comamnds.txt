# Create workspace folder and a src/ inside it
mkdir -p ~/ros2_ws/src
cd ~/ros2_ws/src

# Panda demo (MoveIt2 + RViz + fake ros2_control)
git clone https://github.com/moveit/moveit_resources.git

# RRBot demo (Gazebo + ros2_control)
git clone https://github.com/ros-controls/ros2_control_demos.git

cd ~/ros2_ws
rosdep install --from-paths src --ignore-src -yr

colcon build --symlink-install
source install/setup.bash

ros2 launch moveit_resources_panda_moveit_config demo.launch.py
